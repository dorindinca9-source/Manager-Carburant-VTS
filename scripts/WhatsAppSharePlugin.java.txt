package ro.vts.managercarburant;

import android.content.ClipData;
import android.content.Intent;
import android.net.Uri;
import android.util.Base64;

import androidx.core.content.FileProvider;

import com.getcapacitor.JSArray;
import com.getcapacitor.JSObject;
import com.getcapacitor.Plugin;
import com.getcapacitor.PluginCall;
import com.getcapacitor.PluginMethod;
import com.getcapacitor.annotation.CapacitorPlugin;

import org.json.JSONObject;

import java.io.File;
import java.io.FileOutputStream;
import java.util.ArrayList;

@CapacitorPlugin(name = "WhatsAppShare")
public class WhatsAppSharePlugin extends Plugin {

    @PluginMethod
    public void share(PluginCall call) {
        try {
            JSArray input = call.getArray("files");
            if (input == null || input.length() == 0) {
                call.reject("Nu există fișiere pentru trimitere.");
                return;
            }

            File dir = new File(getContext().getCacheDir(), "shared_exports");
            if (!dir.exists() && !dir.mkdirs()) {
                call.reject("Nu s-a putut crea directorul temporar.");
                return;
            }

            File[] oldFiles = dir.listFiles();
            if (oldFiles != null) {
                for (File f : oldFiles) {
                    try { f.delete(); } catch (Exception ignored) {}
                }
            }

            ArrayList<Uri> uris = new ArrayList<>();
            ArrayList<String> mimes = new ArrayList<>();

            for (int i = 0; i < input.length(); i++) {
                JSONObject obj = input.getJSONObject(i);
                String name = sanitizeFileName(obj.optString("name", "document"));
                String mime = obj.optString("mime", "application/octet-stream");
                String data = obj.optString("data", "");

                if (data.isEmpty()) {
                    call.reject("Fișier gol: " + name);
                    return;
                }

                byte[] bytes = Base64.decode(data, Base64.DEFAULT);
                File out = new File(dir, name);
                try (FileOutputStream fos = new FileOutputStream(out)) {
                    fos.write(bytes);
                }

                Uri uri = FileProvider.getUriForFile(
                    getContext(),
                    getContext().getPackageName() + ".vtsfiles",
                    out
                );
                uris.add(uri);
                mimes.add(mime);
            }

            Intent intent;
            if (uris.size() == 1) {
                intent = new Intent(Intent.ACTION_SEND);
                intent.putExtra(Intent.EXTRA_STREAM, uris.get(0));
            } else {
                intent = new Intent(Intent.ACTION_SEND_MULTIPLE);
                intent.putParcelableArrayListExtra(Intent.EXTRA_STREAM, uris);
            }

            intent.setType(commonMime(mimes));
            intent.addFlags(Intent.FLAG_GRANT_READ_URI_PERMISSION);

            String text = call.getString("text", "");
            String title = call.getString("title", "");
            if (!text.isEmpty()) intent.putExtra(Intent.EXTRA_TEXT, text);
            if (!title.isEmpty()) intent.putExtra(Intent.EXTRA_SUBJECT, title);

            ClipData clip = ClipData.newUri(getContext().getContentResolver(), "Manager Carburant VTS", uris.get(0));
            for (int i = 1; i < uris.size(); i++) clip.addItem(new ClipData.Item(uris.get(i)));
            intent.setClipData(clip);

            String whatsappPackage = findWhatsAppPackage();
            if (whatsappPackage == null) {
                call.reject("WhatsApp nu este instalat pe telefon.");
                return;
            }

            intent.setPackage(whatsappPackage);
            getActivity().startActivity(intent);

            JSObject result = new JSObject();
            result.put("shared", true);
            result.put("package", whatsappPackage);
            result.put("count", uris.size());
            call.resolve(result);

        } catch (Exception e) {
            call.reject("Eroare trimitere WhatsApp: " + e.getMessage(), e);
        }
    }

    private String findWhatsAppPackage() {
        String[] packages = {"com.whatsapp", "com.whatsapp.w4b"};
        for (String pkg : packages) {
            try {
                getContext().getPackageManager().getPackageInfo(pkg, 0);
                return pkg;
            } catch (Exception ignored) {}
        }
        return null;
    }

    private String commonMime(ArrayList<String> mimes) {
        if (mimes.isEmpty()) return "*/*";
        String first = mimes.get(0);
        for (String mime : mimes) {
            if (!first.equals(mime)) return "*/*";
        }
        return first;
    }

    private String sanitizeFileName(String name) {
        String cleaned = name.replaceAll("[\\\\/:*?\"<>|]", "_").trim();
        return cleaned.isEmpty() ? "document" : cleaned;
    }
}
